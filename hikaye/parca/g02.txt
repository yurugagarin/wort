böse :: bö·se (bö-ze) = BEZE | KIZGIN | Sen KIZGIN bir canavar gibi dev bir BEZE pastasını çatalla parçalıyorsun; krema tavana sıçrıyor, burnundan duman çıkıyor, tatlı koku bütün mutfağı sarıyor.
wütend :: wü·tend (vü-tınt) = BÜTÜN | ÖFKELİ | Sen o kadar ÖFKELİsin ki BÜTÜN mahalle kulaklarını tıkıyor; tencereler zıplıyor, ağzından kırmızı alev fışkırıyor, ayağının altında kaldırım çatırdıyor.
zufrieden :: zu·frie·den (tsu-fri-dın) = SU + FERİDUN | MEMNUN | Sen SU kuyusuna eğiliyorsun; içinden FERİDUN amca ıslak sakalıyla fırlayıp sana sıcak baklava uzatıyor; her lokmada MEMNUN bir "mmm" çekiyorsun, şerbet çenenden damlıyor.
enttäuscht :: ent·täuscht (ent-toyşt) = EN + TOYS | HAYAL KIRIKLIĞINA UĞRAMIŞ | Sen EN büyük hediye kutusunu açıyorsun: içinden TOYS yerine paslı bir tencere kapağı şangırdayarak yuvarlanıyor; HAYAL KIRIKLIĞINA UĞRAMIŞsın, gözlerin sulanıyor.
überrascht :: ü·ber·rascht (ü-bır-raşt) = ÜBER + RAŞİT | ŞAŞIRMIŞ | Sen ÜBER taksisine biniyorsun; şoför RAŞİT direksiyonu bırakıp sürpriz pasta çıkarıyor, mumları üflüyor; ŞAŞIRMIŞ halde ağzın açık kalıyor, krema yanağına yapışıyor.
sich konzentrieren :: kon·zen·trie·ren (kon-tsen-tri-rın) = KONT + SEN + TREN | ODAKLANMAK | KONT Drakula "SEN!" diye bağırıp TRENin tavanına çıkıyor; sen sivri dişlerine öyle ODAKLANIYORsun ki gözlerin dışarı fırlıyor, lokomotif düdüğü kulaklarını sağır ediyor.
sich setzen :: set·zen (zet-sın) = SET + SEN | OTURMAK | Sen çay SETinin dev fincanına OTURUYORsun; fincan çatlarken içinden "SEN!" diye bir ses yankılanıyor, sıcak çay popona yayılıyor, porselen gıcırdıyor.
sich anmelden :: an·mel·den (an-mel-dın) = AN + MEL + DİN | KAYDOLMAK | Sen kütüphaneye KAYDOLUYORsun; tam o AN yanına MEL Gibson gelip elindeki DİN kitabını imzalatıyor; mühür sesi tak tak kulaklarını çınlatıyor.
sich abmelden :: ab·mel·den (ap-mel-dın) = APP + MEL + DİN | KAYDINI SİLDİRMEK | Sen telefondaki APP ekranından KAYDINI SİLDİRİYORsun; ekrandan MEL Gibson fırlayıp DİN kitabını yelpaze gibi sallıyor, telefon ısınıp cızırdıyor.
passieren :: pas·sie·ren (pa-si-rın) = PAS + SİREN | OLMAK | Sen maçta PAS veriyorsun; top uçarken SİREN gibi ulumaya başlıyor, stadyum çığlıkla doluyor: ne büyük bir OLAY OLUYOR! Kulakların çınlıyor, çimen kokusu burnuna doluyor.
