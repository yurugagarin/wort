beachten :: be·ach·ten (be-ah-tın) = BE + AH + TEN | DİKKATE ALMAK | Doktor "BE!" diye bağırıp iğneyi saplıyor, sen "AH!" diye zıplıyorsun; TENindeki dev kırmızı lekeyi artık DİKKATE ALMAK zorundasın, leke fosforlu yanıp sönüyor.
sich halten an :: hal·ten an (hal-tın an) = HALDUN + AN | UYMAK | Kırmızı ışıkta megafonlu HALDUN bağırıyor: "Bu AN herkes kurala UYMAK zorunda!" Sen donup kalıyorsun, kulakların çınlıyor, ayakların asfalta yapışıyor.
einhalten :: ein·hal·ten (ayn-hal-tın) = AYNA + HALDUN | UYMAK | AYNAdan fırlayan HALDUN elinde dev bir kronometreyle süreye UYMAK için seni kovalıyor; sen terliklerinle şimşek gibi koşuyorsun, tik tak sesi kulağını deliyor.
verstoßen :: ver·sto·ßen (fer-şto-sın) = VER + İŞ + TOSUN | İHLAL ETMEK | Çimlere basma yasağını İHLAL ETMEK için "VER İŞ!" diye bağırıp çimlere dalıyorsun; arkandan dev bir TOSUN toynaklarını gümbürdeterek kovalıyor.
sich trauen :: trau·en (trau-ın) = TIR + AVIN | CESARET ETMEK | Dev bir TIR AVIN peşine takılmış, ormanda homurdanıyor; önüne çıkıp "dur!" demeye CESARET ETMEK için kollarını açıyorsun, kornası kulaklarını sağır ediyor.
wagen :: wa·gen (va-gın) = VAGON | GÖZE ALMAK | Fırtınada raylarda kayan boş bir VAGONa atlamak için bütün çürükleri GÖZE ALMAK zorundasın; tekerlekler kulağında şakırdıyor, rüzgâr saçlarını yoluyor.
riskieren :: ris·kie·ren (ris-ki-ren) = RİSKİ + REN | RİSKE ATMAK | RİSKİ umursamadan bir REN geyiğine atlıyor, bütün paranı RİSKE ATMAK için buzlu uçuruma koşuyorsun; boynuzlar kulağını sıyırıyor, soğuk rüzgâr yüzünü yakıyor.
gelingen :: ge·lin·gen (ge-lin-gın) = GELİN + GEN | BAŞARILI OLMAK | Laboratuvarda gelinlikli GELİN dev tüpe kendi GENini damlatıyor, deney BAŞARILI OLUYOR; pembe köpük tavana fışkırıp sana çarpıyor, burnuna tatlı bir koku doluyor.
misslingen :: miss·lin·gen (mis-lin-gın) = MİS + LİNK + EN | BAŞARISIZ OLMAK | Sahnede MİS tacını takarken sen telefondaki LİNKe basıp EN güzel pozu yayınlıyorsun; sahne çöküyor, her şey BAŞARISIZ OLUYOR, taç tepene düşüp çınlıyor.
scheitern :: schei·tern (şay-tan) = ŞEYTAN | BAŞARISIZ OLMAK | Sınavda ŞEYTAN kulağına kıkırdayıp kâğıdına mürekkep döküyor; sen BAŞARISIZ OLUYORsun, kâğıt dumanlar çıkararak sönüyor, burnunu yanık kokusu dolduruyor.
aufgeben :: auf·ge·ben (auf-ge-bın) = AF + GİBİ | VAZGEÇMEK | Zirveye tırmanırken dev bir kaya sırtına çökünce "AF!" diye yalvarıyorsun; çanta kurşun GİBİ ağırlaşıyor, onu fırlatıp VAZGEÇMEK zorunda kalıyorsun, kaya gümbürdüyor.
durchhalten :: durch·hal·ten (durh-hal-tın) = DUR + HALDUN | DAYANMAK | Buz kesen soğukta "DUR!" diye bağıran HALDUN seni buzdan heykele çeviriyor; dişlerin takırdıyor ama sen DAYANMAK için burnunu bile kıpırdatmıyorsun.
sich anstrengen :: an·stren·gen (an-ştren-gın) = AN + TREN + GEN | ÇABA HARCAMAK | O AN rayların üstünde dev bir TRENi omuzlarınla itiyorsun, GENlerin bile titriyor; ÇABA HARCAMAK için dişlerini sıkıyorsun, kömür kokusu burnunu yakıyor.
sich steigern :: stei·gern (ştay-gan) = STAY + KAN | ARTMAK | Maratonda "STAY!" diye bağırıyorsun ama KANın coşup hızın durmadan ARTIYOR; kalbin davul gibi gümlüyor, dizlerinden duman çıkıyor.
steigern :: stei·gern (ştay-gan) = STAY + KAN | ARTIRMAK | Doktor "STAY!" diye bağırıp KAN şekerini ARTIRMAK için ağzına dev bir çikolata tıkıyor; sen çikolatayı ısırıyorsun, kakao kokusu odayı dolduruyor, kalbin çarpıyor.
belasten :: be·las·ten (be-las-tın) = BE + LASTİK | YÜKLENMEK | Marangoz "BE!" diye bağırıp sırtına dev bir LASTİK yığını yüklüyor; sen bütün yükü tek başına YÜKLENMEK zorundasın, kauçuk kokusu burnunu yakıyor, dizlerin titriyor.
aufwachen :: auf·wa·chen (auf-va-hın) = AF + VAHİM | UYANMAK | Çalar saat "AF!" diye havlıyor, VAHİM bir gümbürtüyle yatağın altından dev bir kedi fırlayıp yüzünü yalıyor; sen çığlık atarak UYANIYORsun, yastık havada uçuyor.
einschlafen :: ein·schla·fen (ayn-şla-fın) = AYNA + ŞU + LAFIN | UYKUYA DALMAK | Hoca dev bir AYNAnın önünde ŞU LAFIN sonunu getiremiyor; sen UYKUYA DALMAK için başını sıraya gömüyorsun, horultun ayna camını titretiyor.
verschlafen :: ver·schla·fen (fer-şla-fın) = VER + ŞU + LAFIN | UYUYAKALMAK | Toplantıya UYUYAKALdın; patron kapıyı kırıp "VER ŞU LAFIN hesabını!" diye bağırıyor, sen çarşafa dolanıp yataktan düşüyorsun, çalar saat çığlık atıyor.
träumen :: träu·men (troy-mın) = TROY + MEN | RÜYA GÖRMEK | Yatakta RÜYA GÖRMEK için gözlerini kapatınca TROY'un dev tahta atı odana dalıyor; içinden siyah takımlı MENler çıkıp sana çay ikram ediyor, tahta gıcırtısı kulaklarını dolduruyor.
